#!/bin/bash
# 自动备份脚本 · 庖丁 Agent
# 用法：./auto-backup.sh [目标路径]

BACKUP_DIR="${1:-/backup}"
DATE=$(date +%Y%m%d_%H%M%S)

# 创建工作目录
mkdir -p ${BACKUP_DIR}/${DATE}

# 备份关键数据
rsync -av --delete /home/nicola/.openclaw/workspace ${BACKUP_DIR}/${DATE}/

# 清理旧备份 (保留 7 天)
find ${BACKUP_DIR} -type d -mtime +7 -exec rm -rf {} \;

echo "✅ 备份完成：${BACKUP_DIR}/${DATE}"
```

📝 使用说明:
1. 赋予执行权限：`chmod +x auto-backup.sh`
2. 执行备份：`./auto-backup.sh /mnt/backup`
3. 查看备份：`ls -lh /mnt/backup`

⚙️ 定时执行 (每天凌晨 2 点):
