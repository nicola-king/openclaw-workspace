**安装**:
```bash

crawl4ai-doctor
```

**使用示例**:
```python
import asyncio
from crawl4ai import *

async def main():
    async with AsyncWebCrawler() as crawler:
        result = await crawler.arun(
            url="https://example.com",
        )
        print(result.markdown)

asyncio.run(main())
```

**太一集成方案**:
```python
# 太一 Crawl4AI Skill
class Crawl4AI_Skill:
    def __init__(self):
        self.crawler = AsyncWebCrawler()
    
    async def crawl_and_optimize(self, url: str) -> Dict:
        # 爬取内容
        result = await self.crawler.arun(url=url)
        
        # AI 可见性优化
        optimized = self.optimize_for_ai(result.markdown)
        
        # 结构化输出
        return {
            "markdown": optimized,
            "ai_score": self.calculate_ai_score(optimized),
            "recommendations": self.generate_recommendations()
        }
```

```python
# ai_crawler_optimizer.py
class AICrawlerOptimizer:
    """AI 爬虫优化器"""
    
    def __init__(self):
        self.crawler = AsyncWebCrawler()
        self.geo_optimizer = GEEOptimizer()
    
    async def optimize_content(self, url: str) -> Dict:
        # 1. 爬取内容
        result = await self.crawler.arun(url=url)
        
        # 2. GEO 优化
        optimized = self.geo_optimizer.optimize(result.markdown)
        
        # 3. AI 可见性评分
        ai_score = self.calculate_ai_score(optimized)
        
        # 4. 生成建议
        recommendations = self.generate_recommendations(optimized)
        
        return {
            "content": optimized,
            "ai_score": ai_score,
            "recommendations": recommendations,
            "engines": {
                "chatgpt": self.check_chatgpt_visibility(optimized),
                "perplexity": self.check_perplexity_visibility(optimized),
                "google_ai": self.check_google_ai_visibility(optimized)
            }
        }
    
    def calculate_ai_score(self, content: str) -> float:
        """计算 AI 可见性评分"""
        score = 0
        
        # 结构化检查
        if "# " in content:  # 有标题
            score += 20
        if "## " in content:  # 有副标题
            score += 10
        if "|" in content:  # 有表格
            score += 20
        if "- [ ]" in content:  # 有清单
            score += 10
        
        # 核心结论前置
        if "> **" in content[:500]:  # 摘要在前 500 字
            score += 20
        
        # 数据支撑
        if any(c.isdigit() for c in content[:1000]):
            score += 20
        
        return score
```
