### 核心流程

#### 1. 数据采集流程
```python
async def collect_live_data(live_id: str):
    # 1. 从缓存读取
    cached = await redis.get(f"live:{live_id}")
    if cached and not is_expired(cached):
        return cached
    
    # 2. 调用 API 获取
    try:
        data = await wechat_api.get_live_info(live_id)
    except APIError:
        # 3. API 失败，降级到浏览器自动化
        data = await browser_collector.fetch(live_id)
    
    # 4. 数据校验
    if not validate_data(data):
        raise DataError("数据格式异常")
    
    # 5. 缓存并返回
    await redis.setex(f"live:{live_id}", 60, data)
    return data
```

#### 2. 异常检测流程
```python
def detect_anomaly(current_data, history_data):
    # 1. 计算移动平均
    ma = calculate_moving_average(history_data, window=10)
    
    # 2. 计算标准差
    std = calculate_std(history_data, window=10)
    
    # 3. 3-Sigma 检测
    z_score = (current_data - ma) / std
    if abs(z_score) > 3:
        return {
            'is_anomaly': True,
            'type': 'viewer_drop' if z_score < 0 else 'viewer_surge',
            'severity': 'high' if abs(z_score) > 5 else 'medium',
            'suggestion': get_suggestion(z_score)
        }
    
    return {'is_anomaly': False}
```

def generate_suggestion(state, history):
